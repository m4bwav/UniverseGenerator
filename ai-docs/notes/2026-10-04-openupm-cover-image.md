# OpenUPM cover image

`docs/images/openupm-cover.png`, 1200 x 600 (OpenUPM asks for 2:1, 600 x 300 or larger). It draws a real galaxy: seed `my-seed`, `Preset.Epic` with `Shape = GalaxyShape.Barred` and `Arms = 2`, generated by UniverseGenerator 1.0.0 from nuget.org (the Wyvern Galaxy: 300 systems, 428 lanes). Stars are coloured by class, chokepoints are larger, and bridge lanes are orange. The title, a one-line pitch and `Galaxy.Generate("my-seed")` sit on the left.

The OpenUPM form's "Cover image URL" takes a raw GitHub URL. Use the one pinned to the commit that added the image, so the link does not depend on a branch name; it stays reachable because pull requests here merge with merge commits:

https://raw.githubusercontent.com/m4bwav/UniverseGenerator/6e87ad4586ce6e32986bb3619d2cc25e3e8656db/docs/images/openupm-cover.png

Topic: only **Procedural Generation**. OpenUPM treats unrelated topics as spam ([the submission note](2026-10-04-openupm-submission.md)), and Level Design was considered and left out.

## Rebuild

Run the dump with .NET 10 (`dotnet run dump.cs -- my-seed`, which writes `map.csv`), then `python render.py <folder holding map.csv>` (Pillow; Windows fonts Segoe UI and Consolas). The output matches the committed file byte for byte.

`dump.cs`:

```csharp
#:package UniverseGenerator@1.0.0
#:property PublishAot=false
using UniverseGeneration;
var seed = args.Length > 0 ? args[0] : "my-seed";
var g = Galaxy.Generate(seed, Preset.Epic with { Shape = GalaxyShape.Barred, Arms = 2 });
using var w = new StreamWriter("map.csv");
foreach (var m in g.Map) w.WriteLine($"S,{m.X.ToString(System.Globalization.CultureInfo.InvariantCulture)},{m.Y.ToString(System.Globalization.CultureInfo.InvariantCulture)},{m.StarClass},{m.Danger},{(m.Chokepoint?1:0)}");
foreach (var l in g.Lanes) w.WriteLine($"L,{l.A},{l.B},{(l.Bridge?1:0)}");
Console.WriteLine($"{g.Name} {g.Map.Count} systems {g.Lanes.Count} lanes");
```

`render.py`:

```python
import csv, math, random, sys
from PIL import Image, ImageDraw, ImageFilter, ImageFont
S = sys.argv[1]
W, H, K = 1200, 600, 1
pts, lanes = [], []
for r in csv.reader(open(S + "/map.csv")):
    if r[0] == "S": pts.append((float(r[1]), float(r[2]), r[3], int(r[4]), r[5] == "1"))
    else: lanes.append((int(r[1]), int(r[2]), r[3] == "1"))
xs = [p[0] for p in pts]; ys = [p[1] for p in pts]
cx, cy = (min(xs)+max(xs))/2, (min(ys)+max(ys))/2
span = max(max(xs)-min(xs), max(ys)-min(ys))
size = 540; ox, oy = 880, 300
def P(i): x, y = pts[i][0], pts[i][1]; return (ox + (x-cx)/span*size, oy + (y-cy)/span*size)
col = {"O":(155,176,255),"B":(170,191,255),"A":(202,215,255),"F":(248,247,255),"G":(255,244,214),"K":(255,210,161),"M":(255,180,140)}
img = Image.new("RGB", (W, H), (8, 10, 22))
# background gradient + faint stars
d = ImageDraw.Draw(img)
for y in range(H):
    t = y / H; d.line([(0, y), (W, y)], fill=(int(8+6*t), int(10+4*t), int(24+14*t)))
rnd = random.Random(7)
for _ in range(700):
    x, y = rnd.uniform(0, W), rnd.uniform(0, H); b = rnd.randint(40, 120)
    d.point((x, y), fill=(b, b, b+15))
# core glow
glow = Image.new("RGB", (W, H), (0, 0, 0)); gd = ImageDraw.Draw(glow)
for r, c in [(260, (30, 22, 60)), (170, (60, 40, 90)), (90, (120, 90, 120)), (40, (200, 170, 140))]:
    gd.ellipse([ox-r, oy-r*0.9, ox+r, oy+r*0.9], fill=c)
glow = glow.filter(ImageFilter.GaussianBlur(60))
img = Image.blend(img, Image.composite(glow, img, Image.new("L", (W, H), 255)), 0.0)
from PIL import ImageChops
img = ImageChops.add(img, glow)
# lanes
lay = Image.new("RGBA", (W, H), (0, 0, 0, 0)); ld = ImageDraw.Draw(lay)
for a, b, br in lanes:
    ld.line([P(a), P(b)], fill=(255, 160, 90, 150) if br else (120, 160, 230, 70), width=2 if br else 1)
img = Image.alpha_composite(img.convert("RGBA"), lay)
# star halos then cores
halo = Image.new("RGBA", (W, H), (0, 0, 0, 0)); hd = ImageDraw.Draw(halo)
for i, p in enumerate(pts):
    x, y = P(i); c = col.get(p[2], (255, 255, 255)); r = 7 if p[4] else 5
    hd.ellipse([x-r, y-r, x+r, y+r], fill=c + (110,))
halo = halo.filter(ImageFilter.GaussianBlur(3))
img = Image.alpha_composite(img, halo); d = ImageDraw.Draw(img)
for i, p in enumerate(pts):
    x, y = P(i); c = col.get(p[2], (255, 255, 255)); r = 2.6 if p[4] else 1.8
    d.ellipse([x-r, y-r, x+r, y+r], fill=c)
# text
def font(names, s):
    for n in names:
        try: return ImageFont.truetype(n, s)
        except OSError: pass
    return ImageFont.load_default()
f1 = font(["segoeuib.ttf"], 64)
f2 = font(["segoeui.ttf"], 26)
f3 = font(["consola.ttf"], 22)
d.text((64, 170), "Universe", font=f1, fill=(240, 240, 255))
d.text((64, 238), "Generator", font=f1, fill=(240, 240, 255))
d.text((66, 330), "Seeded galaxies, star systems,", font=f2, fill=(185, 195, 225))
d.text((66, 362), "planets and moons for Unity", font=f2, fill=(185, 195, 225))
d.text((66, 420), 'Galaxy.Generate("my-seed")', font=f3, fill=(255, 205, 140))
img.convert("RGB").save(S + "/openupm-cover.png", optimize=True)
```
